function updateClocks() {
    const now = new Date();
    const pad = (n) => String(n).padStart(2,'0');
    const h = pad(now.getHours()), m = pad(now.getMinutes()), s = pad(now.getSeconds());
    const ltEl = document.getElementById('localTime');
    if (ltEl) ltEl.innerHTML = h+':'+m+'<span class="time-seconds">:'+s+'</span>';

    const fmt = (tz) => {
        try {
            return new Intl.DateTimeFormat('en-GB',{timeZone:tz,hour:'2-digit',minute:'2-digit',hour12:false}).format(now);
        } catch(e){ return '—'; }
    };
    const rEl = document.getElementById('tzRome');
    const lEl = document.getElementById('tzLondon');
    const nEl = document.getElementById('tzNY');
    if (rEl) rEl.textContent = fmt('Europe/Rome');
    if (lEl) lEl.textContent = fmt('Europe/London');
    if (nEl) nEl.textContent = fmt('America/New_York');
}
setInterval(updateClocks, 1000);
updateClocks();

function handleKeydown(e) {
    if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) {
        e.preventDefault();
        sendMessage();
    }
}

let toastTimer;
function showToast(msg) {
    const toast = document.getElementById('toast');
    document.getElementById('toastMsg').textContent = msg;
    toast.style.opacity = '1';
    toast.style.transform = 'translateY(0)';
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => {
        toast.style.opacity = '0';
        toast.style.transform = 'translateY(8px)';
    }, 2800);
}

function getAIResponse(question) {
    const q = question.toLowerCase();
    if (q.includes('restaurant') || q.includes('food') || q.includes('eat') || q.includes('café') || q.includes('coffee'))
        return AI_RESPONSES.restaurant[Math.floor(Math.random() * AI_RESPONSES.restaurant.length)];
    if (q.includes('itinerary') || q.includes('day') || q.includes('plan') || q.includes('trip'))
        return AI_RESPONSES.itinerary[Math.floor(Math.random() * AI_RESPONSES.itinerary.length)];
    if (q.includes('transport') || q.includes('airport') || q.includes('metro') || q.includes('bus') || q.includes('taxi'))
        return AI_RESPONSES.transport[0];
    return AI_RESPONSES.default[0];
}

function renderMarkdown(text) {
    return text
        .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
        .replace(/\n\n/g, '</p><p>')
        .replace(/\n/g, '<br>')
        .replace(/^/, '<p>').replace(/$/, '</p>');
}

function addUserMsg(text) {
    const div = document.createElement('div');
    div.className = 'msg-user fade-in';
    div.innerHTML = `
        <div class="msg-avatar"><i class="bi bi-person-fill"></i></div>
        <div class="msg-bubble">${text.replace(/</g,'&lt;').replace(/>/g,'&gt;')}</div>
    `;
    messagesEl.appendChild(div);
    messagesEl.scrollTop = messagesEl.scrollHeight;
}

function addTypingIndicator() {
    const div = document.createElement('div');
    div.className = 'msg-ai fade-in'; div.id = 'typingIndicator';
    div.innerHTML = `
        <div class="msg-avatar"><i class="bi bi-robot"></i></div>
        <div class="msg-typing">
            <div class="typing-dot"></div>
            <div class="typing-dot"></div>
            <div class="typing-dot"></div>
        </div>
    `;
    messagesEl.appendChild(div);
    messagesEl.scrollTop = messagesEl.scrollHeight;
}

function removeTypingIndicator() {
    const el = document.getElementById('typingIndicator');
    if (el) el.remove();
}

function addAIMsg(text) {
    const now = new Date().toLocaleTimeString('en-GB',{hour:'2-digit',minute:'2-digit'});
    const div = document.createElement('div');
    div.className = 'msg-ai fade-in';
    div.innerHTML = `
        <div class="msg-avatar"><i class="bi bi-robot"></i></div>
        <div>
            <div class="msg-bubble">${renderMarkdown(text)}</div>
            <div class="msg-ai-meta"><i class="bi bi-stars me-1"></i>AI Travel Assistant · ${now}</div>
        </div>
    `;
    messagesEl.appendChild(div);
    messagesEl.scrollTop = messagesEl.scrollHeight;
}

async function sendMessage() {
    const text = promptInput.value.trim();
    if (!text || isLoading) return;

    /* Show messages, hide empty state */
    emptyEl.style.display = 'none';
    messagesEl.style.display = 'flex';

    /* Lock UI */
    isLoading = true;
    sendBtn.classList.add('loading');
    sendBtn.disabled = true;

    addUserMsg(text);
    promptInput.value = '';
    promptInput.dispatchEvent(new Event('input'));

    addTypingIndicator();

    /* Simulate AI response delay */
    const delay = 1200 + Math.random() * 800;
    await new Promise(r => setTimeout(r, delay));

    removeTypingIndicator();
    addAIMsg(getAIResponse(text));

    /* Unlock UI */
    isLoading = false;
    sendBtn.classList.remove('loading');
    sendBtn.disabled = false;
    promptInput.focus();
}
