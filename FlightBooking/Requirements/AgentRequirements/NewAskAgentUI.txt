        function renderMarkdown(text) {
            /* Satır temizliği */
            let html = text
                /* Başlıklar */
                .replace(/^### (.+)$/gm, '<h5 style="font-family:var(--font-display);font-size:13px;font-weight:700;margin:10px 0 4px">$1</h5>')
                .replace(/^## (.+)$/gm,  '<h4 style="font-family:var(--font-display);font-size:14px;font-weight:700;margin:12px 0 5px">$1</h4>')
                .replace(/^# (.+)$/gm,   '<h3 style="font-family:var(--font-display);font-size:15px;font-weight:800;margin:14px 0 6px">$1</h3>')
                /* Kalın */
                .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
                /* İtalik */
                .replace(/\*(.*?)\*/g, '<em>$1</em>')
                /* Yatay çizgi */
                .replace(/^---$/gm, '<hr style="border:none;border-top:1px solid var(--slate-200);margin:10px 0;">')
                /* Bullet list */
                .replace(/^[•\-\*] (.+)$/gm, '<li>$1</li>')
                /* Numaralı list */
                .replace(/^\d+\. (.+)$/gm, '<li>$1</li>');

            /* li'leri ul'e sar */
            html = html.replace(/(<li>.*?<\/li>\n?)+/gs, (match) => `<ul style="padding-left:18px;margin:6px 0;">${match}</ul>`);

            /* Paragraf */
            const lines = html.split('\n');
            const result = [];
            for (let i = 0; i < lines.length; i++) {
                const line = lines[i].trim();
                if (!line) { result.push(''); continue; }
                if (line.startsWith('<h') || line.startsWith('<ul') || line.startsWith('<hr')) {
                    result.push(line);
                } else {
                    result.push(`<p style="margin:0 0 6px;">${line}</p>`);
                }
            }
            return result.join('');
        }

        /* condition metnine göre emoji seç */
        function getWeatherIcon(condition) {
            if (!condition) return '🌡️';
            const c = String(condition).toLowerCase();

            if (c.includes('thunder') || c.includes('storm')) return '⛈️';
            if (c.includes('snow') || c.includes('sleet') || c.includes('flurr')) return '❄️';
            if (c.includes('rain') || c.includes('shower') || c.includes('drizzle')) return '🌧️';
            if (c.includes('partly cloudy') || c.includes('partly sunny')) return '🌤️';
            if (c.includes('cloud') || c.includes('overcast')) return '☁️';
            if (c.includes('sunny') || c.includes('clear') || c.includes('fair')) return '☀️';
            return '🌡️';
        }

        /* "5:05 AM" → "05:05" biçimine çevir; parse edilemezse orijinali döndür */
        function formatClock(value) {
            if (!value) return '-';
            const raw = String(value).trim();
            const m = raw.match(/^(\d{1,2}):(\d{2})\s*(AM|PM)?$/i);
            if (!m) return raw;
            let h = parseInt(m[1], 10);
            const min = m[2];
            const suffix = m[3] ? m[3].toUpperCase() : null;
            if (suffix === 'PM' && h < 12) h += 12;
            if (suffix === 'AM' && h === 12) h = 0;
            return String(h).padStart(2, '0') + ':' + min;
        }

        /* Boş/undefined değerler için güvenli gösterim */
        function val(v, suffix) {
            if (v === null || v === undefined || v === '') return '-';
            return escapeHtml(v) + (suffix || '');
        }

        /* ════════════════════════════════════════════
           sendMessage()  —  gerçek API çağrısı
        ════════════════════════════════════════════ */
        async function sendMessage() {
            const text = promptInput.value.trim();
            if (!text || isLoading) return;

            /* Boş durumu gizle, mesajları göster */
            emptyEl.style.display = 'none';
            messagesEl.style.display = 'flex';

            /* UI kilitle */
            isLoading = true;
            sendBtn.classList.add('loading');
            sendBtn.disabled = true;

            /* NOT: Hava durumu kartına burada dokunulmuyor.
               Eski veri varsa yeni cevap gelene kadar ekranda kalır. */

            /* Kullanıcı mesajını ekle */
            addUserMessage(text);
            promptInput.value = '';
            promptInput.dispatchEvent(new Event('input'));

            /* Agent activity başlat */
            showAgentActivity();

            /* Typing indicator */
            showTypingIndicator();

            try {
                const response = await fetch('/Agent/AskAgent', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'RequestVerificationToken': document.querySelector('input[name="__RequestVerificationToken"]')?.value || ''
                    },
                    body: JSON.stringify({ prompt: text })
                });

                /* DEĞİŞTİ: JSON parse hatasına karşı korumalı okuma */
                let data = null;
                try {
                    data = await response.json();
                } catch (parseErr) {
                    console.error('JSON parse hatası:', parseErr);
                }

                removeTypingIndicator();
                completeAgentActivity();

                /* DEĞİŞTİ: spec'e uygun hata kontrolü */
                if (!response.ok || !data || !data.isSuccess) {
                    console.error('Agent API başarısız:', {
                        status: response.status,
                        statusText: response.statusText,
                        payload: data
                    });
                    addAIMessage('Şu anda hava durumu bilgisi alınamadı. Lütfen tekrar deneyin.', null, true);
                    return;
                }

                /* AI cevabı */
                addAIMessage(data.response || 'Yanıt boş döndü.', { model: data.model });

                /* YENİ: Weather intent ise sağ paneli güncelle */
                if (data.intent === 'Weather' && data.weather) {
                    updateWeatherPanel(data.weather);
                }

            } catch (err) {
                removeTypingIndicator();
                completeAgentActivity();
                addAIMessage('Şu anda hava durumu bilgisi alınamadı. Lütfen tekrar deneyin.', null, true);
                console.error('Agent API hatası:', err);
            } finally {
                /* UI aç */
                isLoading = false;
                sendBtn.classList.remove('loading');
                sendBtn.disabled = false;
                promptInput.focus();
                /* Activity paneli 3sn sonra gizle */
                setTimeout(resetAgentActivity, 3000);
            }
        }
