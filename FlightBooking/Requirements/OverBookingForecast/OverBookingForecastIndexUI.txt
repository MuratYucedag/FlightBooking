@{
    ViewData["Title"] = "2027 AI Overbooking Forecast";
    Layout = "~/Areas/Admin/Views/Layout/AdminLayout.cshtml";

    @* ─── Parse: "01.01.2027 | Morning-1 | Tahmini NoShow: 15 | Önerilen Max Satış: 195" ─── *@
    var rows = new List<(string Date, string Slot, int NoShow, int MaxSale)>();
    foreach (var line in Model)
    {
        try
        {
            var p = line.Split('|');
            var date    = p[0].Trim();
            var slot    = p[1].Trim();
            var noshow  = int.Parse(System.Text.RegularExpressions.Regex.Match(p[2], @"\d+").Value);
            var maxsale = int.Parse(System.Text.RegularExpressions.Regex.Match(p[3], @"\d+").Value);
            rows.Add((date, slot, noshow, maxsale));
        }
        catch { rows.Add((line, "—", 0, 0)); }
    }

    @* ─── KPI Hesapları ─── *@
    int    capacity    = 180;
    double avgNoShow   = rows.Count > 0 ? rows.Average(r => r.NoShow)   : 0;
    double avgMaxSale  = rows.Count > 0 ? rows.Average(r => r.MaxSale)  : 0;
    double avgExtra    = rows.Count > 0 ? rows.Average(r => r.MaxSale - capacity) : 0;
    int    totalFlights= rows.Count;

    @* En riskli / en güvenli slot *@
    string riskySlot  = "—";
    string safeSlot   = "—";
    var slotGroups = rows.Where(r => r.Slot != "—")
                         .GroupBy(r => r.Slot)
                         .Select(g => new { Slot = g.Key, Avg = g.Average(x => x.NoShow) })
                         .OrderByDescending(x => x.Avg).ToList();
    if (slotGroups.Any()) { riskySlot = slotGroups.First().Slot; safeSlot = slotGroups.Last().Slot; }

    @* Potansiyel ek gelir (örnek: ekstra koltuk × 120 EUR) *@
    double ticketPrice  = 120.0;
    double totalExtra   = rows.Sum(r => Math.Max(r.MaxSale - capacity, 0));
    double potRevenue   = totalExtra * ticketPrice;

    @* Risk eşikleri *@
    int    highRisk  = 18;
    int    midRisk   = 12;

    Func<int, string> riskLvl   = (n) => n >= highRisk ? "Critical" : n >= midRisk ? "High" : n >= 8 ? "Medium" : "Low";
    Func<int, string> riskLabel = (n) => n >= highRisk ? "Kritik"   : n >= midRisk ? "Yüksek" : n >= 8 ? "Orta" : "Düşük";

    Func<int, string> riskIcon  = (n) => n >= highRisk ? "bi-fire"  : n >= midRisk ? "bi-exclamation-triangle-fill" : n >= 8 ? "bi-dash-circle" : "bi-check-circle";
    Func<int, string> opComment = (n) => n >= highRisk
        ? "Agresif overbooking + standby ekip gerekli."
        : n >= midRisk ? "Kontrollü overbooking uygulanabilir."
        : n >= 8       ? "Standart overbooking yeterli."
        : "Düşük risk, dikkatli satış önerilir.";

function fcFilter(risk, btn) {
    document.querySelectorAll('.fc-chip').forEach(function(c){ c.classList.remove('active'); });
    btn.classList.add('active');
    var rows = document.querySelectorAll('#fcTable tbody .fc-row');
    var vis  = 0;
    rows.forEach(function(row) {
        var show = risk === 'all' || row.dataset.risk === risk;
        row.style.display = show ? '' : 'none';
        if (show) vis++;
    });
    document.getElementById('fcRecCount').textContent = vis + ' kayıt';
}

function fcSearch(val) {
    var rows = document.querySelectorAll('#fcTable tbody .fc-row');
    var vis  = 0;
    val = val.toLowerCase();
    rows.forEach(function(row) {
        var show = val === '' || (row.dataset.search || '').toLowerCase().includes(val);
        row.style.display = show ? '' : 'none';
        if (show) vis++;
    });
    document.getElementById('fcRecCount').textContent = vis + ' kayıt';
}

function exportCSV() {
    var rows  = document.querySelectorAll('#fcTable tbody .fc-row');
    var lines = ['Tarih,Slot,Tahmini NoShow,Max Satış,Ekstra,Risk,Gelir Pot.'];
    rows.forEach(function(row) {
        if (row.style.display === 'none') return;
        var tds = row.querySelectorAll('td');
        var arr = [];
        for (var i = 0; i < 7; i++) arr.push(tds[i] ? tds[i].textContent.trim().replace(/\s+/g,' ') : '');
        lines.push(arr.join(','));
    });
    var blob = new Blob([lines.join('\n')], { type:'text/csv;charset=utf-8;' });
    var url  = URL.createObjectURL(blob);
    var a    = document.createElement('a');
    a.href = url; a.download = 'forecast-ocak2027.csv'; a.click();
    URL.revokeObjectURL(url);
}
